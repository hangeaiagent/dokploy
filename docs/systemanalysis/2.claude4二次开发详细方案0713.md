# GitAgent AI 智能体平台 - 基于 Dokploy 二次开发详细方案

## 1. 项目背景与愿景

### 项目愿景
让开发者能够一键将GitHub代码转化为可立即使用的云应用，无需复杂配置和部署流程

### 核心目标
- 实现GitHub代码库到云服务的分钟级自动化部署
- 内置常用AI能力（大模型API）的自动配置
- 建立按量付费的透明计费体系

## 2. 现有架构分析

### 2.1 Dokploy 现有优势
- **完整的容器化部署系统** - Docker/Docker Swarm 支持
- **多数据库支持** - PostgreSQL, MySQL, Redis, MongoDB, MariaDB
- **Git 集成** - 支持 GitHub, GitLab, Gitea, Bitbucket
- **tRPC 类型安全 API** - 完整的前后端通信架构
- **队列系统** - BullMQ 异步任务处理
- **监控系统** - Go 服务实时监控
- **模块化架构** - 微服务设计，易于扩展

### 2.2 现有架构适配性
```
当前架构 → AI 智能体平台改造
├── apps/dokploy → apps/gitagent (主平台)
├── apps/api → apps/ai-api (AI 服务 API)
├── apps/monitoring → apps/usage-monitor (用量监控)
├── apps/schedules → apps/ai-scheduler (AI 任务调度)
└── packages/server → packages/ai-server (AI 服务层)
```

```typescript
// packages/ai-server/src/services/ai-agent.ts
import { TRPCError } from '@trpc/server';
import { db } from '../db/index';
import { aiAgents, usageRecords } from '../db/schema';
import { z } from 'zod';
import { eq, desc, sql } from 'drizzle-orm';

const createAgentSchema = z.object({
  name: z.string().min(1).max(255),
  description: z.string().optional(),
  githubRepo: z.string().url(),
  framework: z.string().optional(),
  language: z.string().optional(),
  category: z.string().optional(),
  tags: z.array(z.string()).optional(),
});

export const aiAgentService = {
  async createAgent(input: z.infer<typeof createAgentSchema>) {
    const validated = createAgentSchema.parse(input);
    
    // GitHub 仓库信息获取
    const repoInfo = await fetchGitHubRepoInfo(validated.githubRepo);
    
    const agent = await db.insert(aiAgents).values({
      ...validated,
      stars: repoInfo.stars,
      forks: repoInfo.forks,
    }).returning();
    
    return agent[0];
  },

  async getAgentsByCategory(category?: string) {
    const query = db.select().from(aiAgents);
    
    if (category) {
      query.where(eq(aiAgents.category, category));
    }
    
    return query.orderBy(desc(aiAgents.stars)).limit(12);
  },

  async getAgentUsageStats(agentId: string) {
    const stats = await db
      .select({
        totalUsage: sql<number>`sum(${usageRecords.usageAmount})`,
        totalCost: sql<number>`sum(${usageRecords.costAmount})`,
        dailyUsage: sql<number>`sum(case when ${usageRecords.recordedAt} >= now() - interval '1 day' then ${usageRecords.usageAmount} else 0 end)`,
      })
      .from(usageRecords)
      .where(eq(usageRecords.agentId, agentId));
    
    return stats[0];
  },
};

async function fetchGitHubRepoInfo(repoUrl: string) {
  const repoPath = repoUrl.replace('https://github.com/', '');
  const response = await fetch(`https://api.github.com/repos/${repoPath}`);
  const data = await response.json();
  
  return {
    stars: data.stargazers_count || 0,
    forks: data.forks_count || 0,
    language: data.language || 'Unknown',
    description: data.description || '',
  };
}
```

### 5.3 阶段三：代码智能解析引擎（3周）

#### 5.3.1 代码分析服务
```typescript
// packages/ai-server/src/services/code-analyzer.ts
import Parser from 'tree-sitter';
import TypeScript from 'tree-sitter-typescript';
import Python from 'tree-sitter-python';
import Java from 'tree-sitter-java';
import { exec } from 'child_process';
import { promisify } from 'util';
import * as fs from 'fs/promises';
import * as path from 'path';

const execAsync = promisify(exec);

export interface CodeAnalysisResult {
  language: string;
  framework: string;
  dependencies: string[];
  environmentVariables: string[];
  dockerfileContent: string;
  deploymentConfig: any;
  estimatedResources: {
    cpu: string;
    memory: string;
    storage: string;
  };
}

export class CodeAnalyzer {
  private parsers: Map<string, Parser> = new Map();

  constructor() {
    this.initializeParsers();
  }

  private initializeParsers() {
    // TypeScript/JavaScript
    const tsParser = new Parser();
    tsParser.setLanguage(TypeScript.typescript);
    this.parsers.set('typescript', tsParser);

    // Python
    const pyParser = new Parser();
    pyParser.setLanguage(Python);
    this.parsers.set('python', pyParser);

    // Java
    const javaParser = new Parser();
    javaParser.setLanguage(Java);
    this.parsers.set('java', javaParser);
  }

  async analyzeRepository(repoPath: string): Promise<CodeAnalysisResult> {
    const language = await this.detectLanguage(repoPath);
    const framework = await this.detectFramework(repoPath, language);
    const dependencies = await this.extractDependencies(repoPath, language);
    const environmentVariables = await this.extractEnvironmentVariables(repoPath);
    
    const dockerfileContent = await this.generateDockerfile(
      language,
      framework,
      dependencies
    );
    
    const deploymentConfig = await this.generateDeploymentConfig(
      language,
      framework
    );
    
    const estimatedResources = this.estimateResources(language, framework);

    return {
      language,
      framework,
      dependencies,
      environmentVariables,
      dockerfileContent,
      deploymentConfig,
      estimatedResources,
    };
  }

  private async detectLanguage(repoPath: string): Promise<string> {
    const files = await fs.readdir(repoPath);
    
    // 检查关键文件
    if (files.includes('package.json')) return 'javascript';
    if (files.includes('requirements.txt') || files.includes('setup.py')) return 'python';
    if (files.includes('pom.xml') || files.includes('build.gradle')) return 'java';
    if (files.includes('Cargo.toml')) return 'rust';
    if (files.includes('go.mod')) return 'go';
    
    // 通过文件扩展名检测
    const extensions = files.map(f => path.extname(f)).filter(Boolean);
    const extensionCounts = extensions.reduce((acc, ext) => {
      acc[ext] = (acc[ext] || 0) + 1;
      return acc;
    }, {} as Record<string, number>);
    
    const mostCommonExt = Object.entries(extensionCounts)
      .sort(([,a], [,b]) => b - a)[0]?.[0];
    
    const languageMap: Record<string, string> = {
      '.js': 'javascript',
      '.ts': 'typescript',
      '.py': 'python',
      '.java': 'java',
      '.rs': 'rust',
      '.go': 'go',
    };
    
    return languageMap[mostCommonExt] || 'unknown';
  }

  private async detectFramework(repoPath: string, language: string): Promise<string> {
    try {
      switch (language) {
        case 'javascript':
        case 'typescript':
          return await this.detectJSFramework(repoPath);
        case 'python':
          return await this.detectPythonFramework(repoPath);
        case 'java':
          return await this.detectJavaFramework(repoPath);
        default:
          return 'unknown';
      }
    } catch {
      return 'unknown';
    }
  }

  private async detectJSFramework(repoPath: string): Promise<string> {
    const packageJsonPath = path.join(repoPath, 'package.json');
    const packageJson = JSON.parse(await fs.readFile(packageJsonPath, 'utf-8'));
    
    const dependencies = {
      ...packageJson.dependencies,
      ...packageJson.devDependencies,
    };
    
    if (dependencies.next) return 'nextjs';
    if (dependencies.react) return 'react';
    if (dependencies.vue) return 'vue';
    if (dependencies.angular) return 'angular';
    if (dependencies.express) return 'express';
    if (dependencies.fastify) return 'fastify';
    if (dependencies.nuxt) return 'nuxt';
    
    return 'nodejs';
  }

  private async detectPythonFramework(repoPath: string): Promise<string> {
    const requirementsPath = path.join(repoPath, 'requirements.txt');
    let requirements = '';
    
    try {
      requirements = await fs.readFile(requirementsPath, 'utf-8');
    } catch {
      // 尝试 setup.py 或 pyproject.toml
      return 'python';
    }
    
    if (requirements.includes('django')) return 'django';
    if (requirements.includes('flask')) return 'flask';
    if (requirements.includes('fastapi')) return 'fastapi';
    if (requirements.includes('streamlit')) return 'streamlit';
    if (requirements.includes('gradio')) return 'gradio';
    
    return 'python';
  }

  private async detectJavaFramework(repoPath: string): Promise<string> {
    const pomPath = path.join(repoPath, 'pom.xml');
    
    try {
      const pomContent = await fs.readFile(pomPath, 'utf-8');
      
      if (pomContent.includes('spring-boot')) return 'spring-boot';
      if (pomContent.includes('spring')) return 'spring';
      if (pomContent.includes('quarkus')) return 'quarkus';
      if (pomContent.includes('micronaut')) return 'micronaut';
    } catch {
      // 尝试 Gradle
    }
    
    return 'java';
  }

  private async extractDependencies(repoPath: string, language: string): Promise<string[]> {
    switch (language) {
      case 'javascript':
      case 'typescript':
        return await this.extractJSDependencies(repoPath);
      case 'python':
        return await this.extractPythonDependencies(repoPath);
      case 'java':
        return await this.extractJavaDependencies(repoPath);
      default:
        return [];
    }
  }

  private async extractJSDependencies(repoPath: string): Promise<string[]> {
    const packageJsonPath = path.join(repoPath, 'package.json');
    const packageJson = JSON.parse(await fs.readFile(packageJsonPath, 'utf-8'));
    
    return Object.keys({
      ...packageJson.dependencies,
      ...packageJson.devDependencies,
    });
  }

  private async extractPythonDependencies(repoPath: string): Promise<string[]> {
    const requirementsPath = path.join(repoPath, 'requirements.txt');
    
    try {
      const requirements = await fs.readFile(requirementsPath, 'utf-8');
      return requirements
        .split('\n')
        .filter(line => line.trim() && !line.startsWith('#'))
        .map(line => line.split('==')[0].split('>=')[0].split('~=')[0].trim());
    } catch {
      return [];
    }
  }

  private async extractJavaDependencies(repoPath: string): Promise<string[]> {
    const pomPath = path.join(repoPath, 'pom.xml');
    
    try {
      const pomContent = await fs.readFile(pomPath, 'utf-8');
      const dependencyRegex = /<artifactId>(.+?)<\/artifactId>/g;
      const dependencies = [];
      let match;
      
      while ((match = dependencyRegex.exec(pomContent)) !== null) {
        dependencies.push(match[1]);
      }
      
      return dependencies;
    } catch {
      return [];
    }
  }

  private async extractEnvironmentVariables(repoPath: string): Promise<string[]> {
    const envFiles = ['.env', '.env.example', '.env.local'];
    const envVars = new Set<string>();
    
    for (const envFile of envFiles) {
      try {
        const envPath = path.join(repoPath, envFile);
        const envContent = await fs.readFile(envPath, 'utf-8');
        
        envContent.split('\n').forEach(line => {
          const trimmed = line.trim();
          if (trimmed && !trimmed.startsWith('#')) {
            const [key] = trimmed.split('=');
            if (key) {
              envVars.add(key.trim());
            }
          }
        });
      } catch {
        // 文件不存在，跳过
      }
    }
    
    return Array.from(envVars);
  }
```

```typescript
    await db.insert(usageRecords).values({

      serviceType: usage.serviceType,
      usageAmount: usage.usageAmount,
      costAmount: usage.costAmount,
      recordedAt: new Date(),
    });
  }

  async getUserUsage(userId: string, startDate: Date, endDate: Date) {
    const usage = await db
      .select({
        serviceType: usageRecords.serviceType,
        totalUsage: sql<number>`sum(${usageRecords.usageAmount})`,
        totalCost: sql<number>`sum(${usageRecords.costAmount})`,
      })
      .from(usageRecords)
      .where(
        and(
          eq(usageRecords.userId, userId),
          gte(usageRecords.recordedAt, startDate),
          gte(endDate, usageRecords.recordedAt)
        )
      )
      .groupBy(usageRecords.serviceType);

    return usage;
  }

  async getAgentUsage(agentId: string, startDate: Date, endDate: Date) {
    const usage = await db
      .select({
        date: sql<string>`date(${usageRecords.recordedAt})`,
        serviceType: usageRecords.serviceType,
        totalUsage: sql<number>`sum(${usageRecords.usageAmount})`,
        totalCost: sql<number>`sum(${usageRecords.costAmount})`,
      })
      .from(usageRecords)
      .where(
        and(
          eq(usageRecords.agentId, agentId),
          gte(usageRecords.recordedAt, startDate),
          gte(endDate, usageRecords.recordedAt)
        )
      )
      .groupBy(sql`date(${usageRecords.recordedAt})`, usageRecords.serviceType)
      .orderBy(sql`date(${usageRecords.recordedAt})`);

    return usage;
  }

  async predictMonthlyCost(userId: string): Promise<number> {
    // 基于过去7天的使用情况预测月度成本
    const sevenDaysAgo = new Date();
    sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 7);

    const recentUsage = await db
      .select({
        dailyAverage: sql<number>`avg(daily_cost)`,
      })
      .from(
        db
          .select({
            dailyCost: sql<number>`sum(${usageRecords.costAmount}) as daily_cost`,
          })
          .from(usageRecords)
          .where(
            and(
              eq(usageRecords.userId, userId),
              gte(usageRecords.recordedAt, sevenDaysAgo)
            )
          )
          .groupBy(sql`date(${usageRecords.recordedAt})`)
          .as('daily_usage')
      );

    const dailyAverage = recentUsage[0]?.dailyAverage || 0;
    return dailyAverage * 30; // 预测月度成本
  }

  async getTopUsers(limit: number = 10) {
    const topUsers = await db
      .select({
        userId: usageRecords.userId,
        totalCost: sql<number>`sum(${usageRecords.costAmount})`,
        totalUsage: sql<number>`sum(${usageRecords.usageAmount})`,
      })
      .from(usageRecords)
      .groupBy(usageRecords.userId)
      .orderBy(sql`sum(${usageRecords.costAmount}) desc`)
      .limit(limit);

    return topUsers;
  }
```
