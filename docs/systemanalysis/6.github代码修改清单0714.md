```typescript
import * as fs from 'fs/promises';
import * as path from 'path';
import { glob } from 'glob';

export interface CodeAnalysisResult {
  detectedLanguage: string;
  detectedFramework: string;
  buildType: 'nixpacks' | 'dockerfile' | 'buildpacks';
  buildConfig: {
    startCommand?: string;
    buildCommand?: string;
    installCommand?: string;
    buildPath: string;
    port: number;
    packageManager?: string;
    runtimeVersion?: string;
  };
  environmentVariables: EnvironmentVariable[];
  resourceRequirements: {
    cpu: string;
    memory: string;
    storage: string;
    estimatedCost: number;
  };
  securityAnalysis: {
    vulnerabilities: SecurityVulnerability[];
    recommendations: string[];
    riskLevel: 'low' | 'medium' | 'high';
  };
  deploymentReadiness: {
    score: number;
    blockers: string[];
    warnings: string[];
    autoFixAvailable: boolean;
  };
}

export class EnhancedCodeAnalyzer {
  private readonly MAX_ANALYSIS_TIME = 180000; // 3分钟

  async analyzeProject(repoPath: string): Promise<CodeAnalysisResult> {
    const startTime = Date.now();
    
    try {
      // 并行执行多种分析
      const [
        languageAnalysis,
        securityAnalysis,
        deploymentAnalysis
      ] = await Promise.all([
        this.analyzeLanguageAndFramework(repoPath),
        this.performSecurityAnalysis(repoPath),
        this.assessDeploymentReadiness(repoPath)
      ]);

      return {
        ...languageAnalysis,
        securityAnalysis,
        deploymentReadiness: deploymentAnalysis,
      };
    } catch (error) {
      throw new Error(`Code analysis failed: ${error.message}`);
    }
  }

  private async analyzeNodeProject(repoPath: string): Promise<Partial<CodeAnalysisResult>> {
    const packageJsonPath = path.join(repoPath, 'package.json');
    const packageJson = JSON.parse(await fs.readFile(packageJsonPath, 'utf-8'));
    
    const dependencies = { ...packageJson.dependencies, ...packageJson.devDependencies };
    const framework = this.detectNodeFramework(dependencies);
    const packageManager = await this.detectPackageManager(repoPath);
    
    return {
      detectedLanguage: 'javascript',
      detectedFramework: framework,
      buildType: 'nixpacks',
      buildConfig: {
        startCommand: packageJson.scripts?.start || 'npm start',
        buildCommand: packageJson.scripts?.build,
        installCommand: `${packageManager} install`,
        buildPath: '/',
        port: await this.detectPort(repoPath, 3000),
        packageManager,
        runtimeVersion: packageJson.engines?.node || '18',
      },
      environmentVariables: await this.extractEnvironmentVariables(repoPath),
      resourceRequirements: this.estimateResourceRequirements(framework, dependencies),
    };
  }

  private detectNodeFramework(dependencies: Record<string, string>): string {
    const frameworks = [
      { deps: ['next'], name: 'Next.js', priority: 1 },
      { deps: ['@nuxt/core', 'nuxt'], name: 'Nuxt.js', priority: 1 },
      { deps: ['@sveltejs/kit'], name: 'SvelteKit', priority: 1 },
      { deps: ['react'], name: 'React', priority: 3 },
      { deps: ['express'], name: 'Express', priority: 4 },
    ];

    const detected = frameworks
      .filter(({ deps }) => deps.some(dep => dependencies[dep]))
      .sort((a, b) => a.priority - b.priority);

    return detected[0]?.name || 'Node.js';
  }

  private async performSecurityAnalysis(repoPath: string): Promise<any> {
    const vulnerabilities: SecurityVulnerability[] = [];
    const recommendations: string[] = [];
    
    // 扫描敏感信息
    await this.scanForSecrets(repoPath, vulnerabilities);
    
    return {
      vulnerabilities,
      recommendations,
      riskLevel: this.calculateRiskLevel(vulnerabilities),
    };
  }

  private async scanForSecrets(repoPath: string, issues: SecurityVulnerability[]): Promise<void> {
    const secretPatterns = [
      { pattern: /sk-[a-zA-Z0-9]{48}/, type: 'OpenAI API Key' },
      { pattern: /AIza[0-9A-Za-z\\-_]{35}/, type: 'Google API Key' },
      { pattern: /AKIA[0-9A-Z]{16}/, type: 'AWS Access Key' },
    ];

    try {
      const files = await glob('**/*.{js,ts,py,json}', {
        cwd: repoPath,
        ignore: ['node_modules/**', '.git/**'],
      });

      for (const file of files.slice(0, 100)) {
        const content = await fs.readFile(path.join(repoPath, file), 'utf-8');
        
        for (const { pattern, type } of secretPatterns) {
          if (pattern.test(content)) {
            issues.push({
              type: 'secret',
              severity: 'high',
              description: `Potential ${type} found in source code`,
              file,
              recommendation: `Remove ${type} from source code`,
              autoFixAvailable: false,
            });
          }
        }
      }
    } catch (error) {
      console.warn('Secret scanning failed:', error);
    }
  }
}
```

### 2. 项目导入服务

**文件**: `packages/server/src/services/project-ingestion.ts`

```typescript
import { GithubService } from './github';
import { EnhancedCodeAnalyzer } from './enhanced-code-analyzer';
import { ApplicationService } from './application';
import { db } from '../db';
import { userDeployedProjects } from '../db/schema';
import { randomBytes, createHash } from 'crypto';

export class ProjectIngestionService {
  constructor(
    private githubService: GithubService,
    private codeAnalyzer: EnhancedCodeAnalyzer,
    private applicationService: ApplicationService
  ) {}

  async ingestFromGithub(
    userId: string,
    githubUrl: string,
    projectId: string,
    options: ProjectIngestionOptions = {}
  ): Promise<{
    application: any;
    analysisResult: any;
    apiKey: string;
  }> {
    let repoPath: string | undefined;
    let createdAppId: string | undefined;

    try {
      // 1. 验证GitHub URL
      await this.validateGithubUrl(githubUrl);

      // 2. 克隆仓库
      repoPath = await this.githubService.cloneRepository(githubUrl);

      // 3. 代码分析
      const analysisResult = await this.codeAnalyzer.analyzeProject(repoPath);

      // 4. 检查部署就绪性
      if (analysisResult.deploymentReadiness.score < 70) {
        throw new Error('Project readiness score too low');
      }

      // 5. 创建应用配置
      const appConfig = this.buildApplicationConfig(
        githubUrl,
        analysisResult,
        projectId,
        options
      );

      // 6. 创建Dokploy应用
      const application = await this.applicationService.createApplication(
        appConfig,
        userId
      );
      createdAppId = application.id;

      // 7. 生成API密钥
      const apiKey = this.generateApiKey();

      // 8. 记录部署项目
      await db.insert(userDeployedProjects).values({
        userId,
        dokployApplicationId: application.id,
        githubUrl,
        apiKey: this.hashApiKey(apiKey),
        analysisCache: analysisResult,
        deploymentStatus: 'ready',
      });

      return {
        application,
        analysisResult,
        apiKey,
      };

    } catch (error) {
      await this.rollbackIngestion(createdAppId);
      throw error;
    } finally {
      if (repoPath) {
        await fs.rm(repoPath, { recursive: true, force: true }).catch(console.warn);
      }
    }
  }

  private async validateGithubUrl(githubUrl: string): Promise<void> {
    const urlPattern = /^https:\/\/github\.com\/[\w-]+\/[\w-]+(?:\.git)?$/;
    if (!urlPattern.test(githubUrl)) {
      throw new Error('Invalid GitHub URL format');
    }

    const repoPath = githubUrl.replace('https://github.com/', '').replace('.git', '');
    const response = await fetch(`https://api.github.com/repos/${repoPath}`);
    
    if (!response.ok) {
      throw new Error(`Repository not accessible: ${response.status}`);
    }
  }

  private generateApiKey(): string {
    return 'gta_' + randomBytes(32).toString('hex');
  }

  private hashApiKey(apiKey: string): string {
    return createHash('sha256').update(apiKey).digest('hex');
  }
}
```

### 3. AI代理服务

**文件**: `packages/server/src/services/ai-proxy.ts`

```typescript
import { db } from '../db';
import { userAiCredentials, userDeployedProjects, billingRecords } from '../db/schema';
import { eq, and } from 'drizzle-orm';
import { createCipheriv, createDecipheriv, scryptSync, randomBytes } from 'crypto';

export class AIProxyService {
  private readonly encryptionKey = scryptSync(process.env.AI_ENCRYPTION_KEY!, 'salt', 32);

  async proxyRequest(
    apiKey: string,
    provider: string,
    payload: any
  ): Promise<any> {
    try {
      // 1. 验证API密钥
      const project = await this.validateApiKey(apiKey);
      
      // 2. 获取用户AI凭证
      const userApiKey = await this.getUserApiKey(project.userId, provider);
      
      // 3. 转发请求
      const response = await this.forwardRequest(userApiKey, provider, payload);
      
      // 4. 记录使用量
      await this.recordUsage(project, provider, payload, response);

      return response;
    } catch (error) {
      throw error;
    }
  }

  private async validateApiKey(apiKey: string): Promise<any> {
    const hashedKey = require('crypto').createHash('sha256').update(apiKey).digest('hex');
    
    const project = await db.query.userDeployedProjects.findFirst({
      where: eq(userDeployedProjects.apiKey, hashedKey),
    });

    if (!project) {
      throw new Error('Invalid API key');
    }

    return project;
  }

  private async getUserApiKey(userId: string, provider: string): Promise<string> {
    const creds = await db.query.userAiCredentials.findFirst({
      where: and(
        eq(userAiCredentials.userId, userId),
        eq(userAiCredentials.provider, provider),
        eq(userAiCredentials.isActive, true)
      ),
    });

    if (!creds) {
      throw new Error(`No ${provider} credentials configured`);
    }

    return this.decrypt(creds.encryptedApiKey);
  }

  private async forwardRequest(apiKey: string, provider: string, payload: any): Promise<any> {
    const endpoints = {
      openai: 'https://api.openai.com/v1/chat/completions',
      anthropic: 'https://api.anthropic.com/v1/messages',
    };

    const endpoint = endpoints[provider];
    if (!endpoint) {
      throw new Error(`Unsupported provider: ${provider}`);
    }

    const response = await fetch(endpoint, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${apiKey}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify(payload),
    });

    return await response.json();
  }

  private async recordUsage(project: any, provider: string, payload: any, response: any): Promise<void> {
    const usage = response.usage;
    if (!usage) return;

    const pricing = this.getPricing(provider, payload.model);
    const inputCost = (usage.prompt_tokens / 1000) * pricing.input;
    const outputCost = (usage.completion_tokens / 1000) * pricing.output;

    await db.insert(billingRecords).values([
      {
        userId: project.userId,
        deployedProjectId: project.id,
        serviceType: 'AI_TOKENS_INPUT',
        quantity: usage.prompt_tokens.toString(),
        unitPrice: pricing.input.toString(),
        totalCost: inputCost.toString(),
      },
      {
        userId: project.userId,
        deployedProjectId: project.id,
        serviceType: 'AI_TOKENS_OUTPUT',
        quantity: usage.completion_tokens.toString(),
        unitPrice: pricing.output.toString(),
        totalCost: outputCost.toString(),
      },
    ]);
  }

  private encrypt(text: string): string {
    const iv = randomBytes(16);
    const cipher = createCipheriv('aes-256-cbc', this.encryptionKey, iv);
    let encrypted = cipher.update(text, 'utf8', 'hex');
    encrypted += cipher.final('hex');
    return iv.toString('hex') + ':' + encrypted;
  }

  private decrypt(encryptedText: string): string {
    const [ivHex, encrypted] = encryptedText.split(':');
    const iv = Buffer.from(ivHex, 'hex');
    const decipher = createDecipheriv('aes-256-cbc', this.encryptionKey, iv);
    let decrypted = decipher.update(encrypted, 'hex', 'utf8');
    decrypted += decipher.final('utf8');
    return decrypted;
  }

  private getPricing(provider: string, model: string): { input: number; output: number } {
    const pricing = {
      openai: {
        'gpt-4': { input: 0.03, output: 0.06 },
        'gpt-3.5-turbo': { input: 0.001, output: 0.002 },
      },
      anthropic: {
        'claude-3-opus': { input: 0.015, output: 0.075 },
      },
    };

    return pricing[provider]?.[model] || { input: 0.001, output: 0.002 };
  }
}
```

```typescript
import { simpleGit } from 'simple-git';
import { nanoid } from 'nanoid';
import * as fs from 'fs/promises';
import * as os from 'os';
import * as path from 'path';

export class GithubService {
  // ... 现有方法 ...

  /**
   * 克隆公共仓库到临时目录
   */
  async cloneRepository(repoUrl: string): Promise<string> {
    const tempDir = await fs.mkdtemp(path.join(os.tmpdir(), `gitagent-${nanoid(6)}-`));
    const git = simpleGit();
    
    try {
      await git.clone(repoUrl, tempDir, ['--depth=1']);
      return tempDir;
    } catch (error) {
      await fs.rm(tempDir, { recursive: true, force: true });
      throw new Error(`Failed to clone repository: ${repoUrl}`);
    }
  }

  /**

  async getRepositoryInfo(repoUrl: string): Promise<{

    description: string;

    forks: number;

  }> {

      description: data.description || '',
      stars: data.stargazers_count || 0,
      forks: data.forks_count || 0,
      language: data.language || 'Unknown',
```

```typescript
import { z } from 'zod';
import { createTRPCRouter, protectedProcedure } from '@/server/api/trpc';
import { AIProxyService } from '@dokploy/server/services/ai-proxy';
import { db } from '@/server/db';
import { userAiCredentials, billingRecords } from '@dokploy/server/db/schema';
import { eq, and, gte, sql } from 'drizzle-orm';

const aiProxyService = new AIProxyService();

export const aiProxyRouter = createTRPCRouter({
  // 设置AI凭证
  setAiCredential: protectedProcedure
    .input(z.object({
      provider: z.enum(['openai', 'anthropic', 'cohere']),
      apiKey: z.string(),
    }))
    .mutation(async ({ ctx, input }) => {
      const encrypted = aiProxyService.encrypt(input.apiKey);
      const keyHash = require('crypto').createHash('sha256').update(input.apiKey).digest('hex');

      await db.insert(userAiCredentials).values({
        userId: ctx.session.user.id,
        provider: input.provider,
        encryptedApiKey: encrypted,
        keyHash,
      }).onConflictDoUpdate({
        target: [userAiCredentials.userId, userAiCredentials.provider],
        set: {
          encryptedApiKey: encrypted,
          keyHash,
          isActive: true,
        },
      });

      return { success: true };
    }),

  // 获取用户AI凭证状态
  getAiCredentials: protectedProcedure
    .query(async ({ ctx }) => {
      const credentials = await db.query.userAiCredentials.findMany({
        where: eq(userAiCredentials.userId, ctx.session.user.id),
        columns: {
          id: true,
          provider: true,
          isActive: true,
          createdAt: true,
          lastUsedAt: true,
        },
      });

      return credentials;
    }),

  // 获取用量统计
  getUsageStats: protectedProcedure
    .input(z.object({
      startDate: z.date(),
      endDate: z.date(),
      serviceType: z.string().optional(),
    }))
    .query(async ({ ctx, input }) => {
      const stats = await db
        .select({
          serviceType: billingRecords.serviceType,
          totalCost: sql<number>`sum(${billingRecords.totalCost})`,
          totalQuantity: sql<number>`sum(${billingRecords.quantity})`,
          recordCount: sql<number>`count(*)`,
        })
        .from(billingRecords)
        .where(
          and(
            eq(billingRecords.userId, ctx.session.user.id),
            gte(billingRecords.recordedAt, input.startDate),
            gte(input.endDate, billingRecords.recordedAt),
            input.serviceType ? eq(billingRecords.serviceType, input.serviceType) : undefined
          )
        )
        .groupBy(billingRecords.serviceType);

      return stats;
    }),

  // 代理AI请求
  proxyAiRequest: protectedProcedure
    .input(z.object({
      apiKey: z.string(),
      provider: z.enum(['openai', 'anthropic', 'cohere']),
      payload: z.any(),
    }))
    .mutation(async ({ input, ctx }) => {
      return await aiProxyService.proxyRequest(
        input.apiKey,
        input.provider,
        input.payload
      );
    }),
```

```typescript
import { Redis } from 'ioredis';

export class RateLimiter {
  private redis: Redis;

  constructor() {
    this.redis = new Redis(process.env.REDIS_URL!);
  }

  async isAllowed(key: string, limit: number, window: number): Promise<boolean> {
    const current = await this.redis.get(key);
    
    if (!current) {
      await this.redis.setex(key, window, 1);
      return true;
    }

    const count = parseInt(current);
    if (count >= limit) {
      return false;
    }

    await this.redis.incr(key);
    return true;
  }

  async getRemainingRequests(key: string, limit: number): Promise<number> {
    const current = await this.redis.get(key);
    const count = current ? parseInt(current) : 0;
    return Math.max(0, limit - count);
  }
}
```

```typescript
    'STORAGE_GB_HOURS': 0.001,
```
