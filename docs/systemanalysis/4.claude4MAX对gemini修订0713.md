### 2. 增强的代码分析服务

**问题：** 原方案的代码分析过于简单，缺少深度分析能力。

**修正方案：**

```typescript
// packages/server/src/services/code-analyzer.service.ts (增强版本)
import * as fs from 'fs/promises';
import * as path from 'path';
import { glob } from 'glob';
import { promisify } from 'util';
import { exec } from 'child_process';

const execAsync = promisify(exec);

export interface DetailedAnalysisResult {
  detectedLanguage: string;
  detectedFramework: string;
  buildPack: 'nixpacks' | 'dockerfile' | 'buildpacks';
  buildPackConfig: Record<string, any>;
  startCommand?: string;
  installCommand?: string;
  buildCommand?: string;
  buildPath: string;
  port: number;
  environmentVariables: EnvironmentVariable[];
  dependencies: Dependency[];
  securityIssues: SecurityIssue[];
  performanceRecommendations: string[];
  estimatedResourceRequirements: ResourceRequirements;
  confidenceScore: number; // 0-100, 分析结果的置信度
}

interface EnvironmentVariable {
  name: string;
  description?: string;
  required: boolean;
  defaultValue?: string;
  sensitive: boolean;
}

interface Dependency {
  name: string;
  version?: string;
  type: 'runtime' | 'build' | 'dev';
  vulnerabilities?: SecurityVulnerability[];
}

interface SecurityIssue {
  type: 'vulnerability' | 'secret' | 'permission';
  severity: 'low' | 'medium' | 'high' | 'critical';
  description: string;
  file?: string;
  line?: number;
  recommendation: string;
}

interface ResourceRequirements {
  cpu: string;
  memory: string;
  storage: string;
  networkBandwidth?: string;
}

export class EnhancedCodeAnalyzerService {
  private readonly MAX_FILE_SIZE = 10 * 1024 * 1024; // 10MB
  private readonly ANALYSIS_TIMEOUT = 300000; // 5 minutes

  async analyze(repoPath: string): Promise<DetailedAnalysisResult> {
    const analysisStart = Date.now();
    
    try {
      // 并行执行多种分析
      const [
        basicAnalysis,
        securityAnalysis,
        dependencyAnalysis,
        performanceAnalysis
      ] = await Promise.all([
        this.performBasicAnalysis(repoPath),
        this.performSecurityAnalysis(repoPath),
        this.performDependencyAnalysis(repoPath),
        this.performPerformanceAnalysis(repoPath)
      ]);

      const result: DetailedAnalysisResult = {
        ...basicAnalysis,
        securityIssues: securityAnalysis,
        dependencies: dependencyAnalysis,
        performanceRecommendations: performanceAnalysis,
        confidenceScore: this.calculateConfidenceScore(basicAnalysis, repoPath),
      };

      const analysisTime = Date.now() - analysisStart;
      console.log(`Code analysis completed in ${analysisTime}ms`);

      return result;
    } catch (error) {
      console.error('Code analysis failed:', error);
      throw new Error(`Code analysis failed: ${error.message}`);
    }
  }

  private async performBasicAnalysis(repoPath: string): Promise<Partial<DetailedAnalysisResult>> {
    // 检查 Dockerfile
    const dockerfilePath = path.join(repoPath, 'Dockerfile');
    if (await this.pathExists(dockerfilePath)) {
      return await this.analyzeDockerfile(dockerfilePath);
    }

    // 检查各种语言的项目配置文件
    const languageAnalyzers = [
      { check: 'package.json', analyzer: this.analyzeNodeProject },
      { check: 'requirements.txt', analyzer: this.analyzePythonProject },
      { check: 'pom.xml', analyzer: this.analyzeJavaProject },
      { check: 'go.mod', analyzer: this.analyzeGoProject },
      { check: 'Cargo.toml', analyzer: this.analyzeRustProject },
      { check: 'composer.json', analyzer: this.analyzePHPProject },
    ];

    for (const { check, analyzer } of languageAnalyzers) {
      if (await this.pathExists(path.join(repoPath, check))) {
        return await analyzer.call(this, repoPath);
      }
    }

    // 如果没有识别到具体框架，尝试通过文件扩展名分析
    return await this.analyzeByFileExtensions(repoPath);
  }

  private async analyzeNodeProject(repoPath: string): Promise<Partial<DetailedAnalysisResult>> {
    const packageJsonPath = path.join(repoPath, 'package.json');
    const packageJson = JSON.parse(await fs.readFile(packageJsonPath, 'utf-8'));
    
    const deps = { ...packageJson.dependencies, ...packageJson.devDependencies };
    const framework = this.detectNodeFramework(deps);
    const port = await this.detectPort(repoPath, framework);
    
    // 检查是否有 TypeScript
    const hasTypeScript = deps.typescript || deps['@types/node'] || 
                         await this.pathExists(path.join(repoPath, 'tsconfig.json'));

    // 检查包管理器
    const packageManager = await this.detectPackageManager(repoPath);

    return {
      detectedLanguage: hasTypeScript ? 'typescript' : 'javascript',
      detectedFramework: framework,
      buildPack: 'nixpacks',
      buildPackConfig: {
        packageManager,
        nodeVersion: packageJson.engines?.node || '18',
      },
      startCommand: packageJson.scripts?.start,
      buildCommand: packageJson.scripts?.build,
      installCommand: `${packageManager} install`,
      buildPath: '/',
      port,
      environmentVariables: await this.extractEnvVariables(repoPath),
      estimatedResourceRequirements: this.estimateResourceRequirements(framework, 'node'),
    };
  }

  private async analyzePythonProject(repoPath: string): Promise<Partial<DetailedAnalysisResult>> {
    const framework = await this.detectPythonFramework(repoPath);
    const pythonVersion = await this.detectPythonVersion(repoPath);
    
    return {
      detectedLanguage: 'python',
      detectedFramework: framework,
      buildPack: 'nixpacks',
      buildPackConfig: {
        pythonVersion,
      },
      startCommand: this.getPythonStartCommand(framework),
      buildPath: '/',
      port: this.getPythonDefaultPort(framework),
      environmentVariables: await this.extractEnvVariables(repoPath),
      estimatedResourceRequirements: this.estimateResourceRequirements(framework, 'python'),
    };
  }

  // 安全分析
  private async performSecurityAnalysis(repoPath: string): Promise<SecurityIssue[]> {
    const issues: SecurityIssue[] = [];
    
    // 检查敏感信息泄露
    await this.scanForSecrets(repoPath, issues);
    
    // 检查危险的文件权限
    await this.checkFilePermissions(repoPath, issues);
    
    // 检查已知的安全配置问题
    await this.checkSecurityConfigurations(repoPath, issues);
    
    return issues;
  }

  private async scanForSecrets(repoPath: string, issues: SecurityIssue[]): Promise<void> {
    const secretPatterns = [
      { pattern: /sk-[a-zA-Z0-9]{48}/, type: 'OpenAI API Key' },
      { pattern: /AIza[0-9A-Za-z\\-_]{35}/, type: 'Google API Key' },
      { pattern: /AKIA[0-9A-Z]{16}/, type: 'AWS Access Key' },
      { pattern: /-----BEGIN PRIVATE KEY-----/, type: 'Private Key' },
      { pattern: /password\s*[:=]\s*['""][^'""]+['""]/, type: 'Hardcoded Password' },
    ];

    try {
      const files = await glob('**/*.{js,ts,py,java,go,rs,php,json,yaml,yml}', {
        cwd: repoPath,
        ignore: ['node_modules/**', '.git/**', 'vendor/**'],
      });

      for (const file of files.slice(0, 100)) { // 限制检查文件数量
        const filePath = path.join(repoPath, file);
        const stats = await fs.stat(filePath);
        
        if (stats.size > this.MAX_FILE_SIZE) continue;
        
        const content = await fs.readFile(filePath, 'utf-8');
        
        for (const { pattern, type } of secretPatterns) {
          const matches = content.match(pattern);
          if (matches) {
            issues.push({
              type: 'secret',
              severity: 'high',
              description: `Potential ${type} found in source code`,
              file,
              recommendation: `Remove ${type} from source code and use environment variables`,
            });
          }
        }
      }
    } catch (error) {
      console.warn('Secret scanning failed:', error);
    }
  }

  private async performDependencyAnalysis(repoPath: string): Promise<Dependency[]> {
    const dependencies: Dependency[] = [];
    
    // Node.js 依赖分析
    if (await this.pathExists(path.join(repoPath, 'package.json'))) {
      const nodeDeps = await this.analyzeNodeDependencies(repoPath);
      dependencies.push(...nodeDeps);
    }
    
    // Python 依赖分析
    if (await this.pathExists(path.join(repoPath, 'requirements.txt'))) {
      const pythonDeps = await this.analyzePythonDependencies(repoPath);
      dependencies.push(...pythonDeps);
    }
    
    return dependencies;
  }

  private calculateConfidenceScore(analysis: Partial<DetailedAnalysisResult>, repoPath: string): number {
    let score = 50; // 基础分数
    
    // 如果检测到具体框架，增加置信度
    if (analysis.detectedFramework && analysis.detectedFramework !== 'unknown') {
      score += 20;
    }
    
    // 如果有启动命令，增加置信度
    if (analysis.startCommand) {
      score += 15;
    }
    
    // 如果有构建配置，增加置信度
    if (analysis.buildCommand) {
      score += 10;
    }
    
    // 如果检测到环境变量，增加置信度
    if (analysis.environmentVariables && analysis.environmentVariables.length > 0) {
      score += 5;
    }
    
    return Math.min(score, 100);
  }

  // 性能分析
  private async performPerformanceAnalysis(repoPath: string): Promise<string[]> {
    const recommendations: string[] = [];
    
    // 检查 Docker 镜像优化
    const dockerfilePath = path.join(repoPath, 'Dockerfile');
    if (await this.pathExists(dockerfilePath)) {
      const content = await fs.readFile(dockerfilePath, 'utf-8');
      
      if (!content.includes('FROM node:') || !content.includes('-alpine')) {
        recommendations.push('Consider using Alpine-based images to reduce image size');
      }
      
      if (!content.includes('USER ')) {
        recommendations.push('Run containers as non-root user for security');
      }
    }
    
    // 检查 Node.js 项目的性能配置
    if (await this.pathExists(path.join(repoPath, 'package.json'))) {
      const pkg = JSON.parse(await fs.readFile(path.join(repoPath, 'package.json'), 'utf-8'));
      
      if (!pkg.scripts?.build) {
        recommendations.push('Add build script for production optimization');
      }
    }
    
    return recommendations;
  }

  // 工具方法
  private async pathExists(filePath: string): Promise<boolean> {
    try {
      await fs.access(filePath);
      return true;
    } catch {
      return false;
    }
  }

  private detectNodeFramework(dependencies: Record<string, string>): string {
    const frameworks = [
      { deps: ['next'], name: 'Next.js' },
      { deps: ['@nuxt/core', 'nuxt'], name: 'Nuxt.js' },
      { deps: ['@sveltejs/kit'], name: 'SvelteKit' },
      { deps: ['@angular/core'], name: 'Angular' },
      { deps: ['react'], name: 'React' },
      { deps: ['vue'], name: 'Vue.js' },
      { deps: ['express'], name: 'Express' },
      { deps: ['fastify'], name: 'Fastify' },
      { deps: ['koa'], name: 'Koa' },
    ];

    for (const { deps, name } of frameworks) {
      if (deps.some(dep => dependencies[dep])) {
        return name;
      }
    }

    return 'Node.js';
  }

  private async detectPackageManager(repoPath: string): Promise<string> {
    if (await this.pathExists(path.join(repoPath, 'pnpm-lock.yaml'))) return 'pnpm';
    if (await this.pathExists(path.join(repoPath, 'yarn.lock'))) return 'yarn';
    return 'npm';
  }

  private async detectPort(repoPath: string, framework: string): Promise<number> {
    // 首先检查环境变量文件
    const envFiles = ['.env', '.env.local', '.env.example'];
    for (const envFile of envFiles) {
      const envPath = path.join(repoPath, envFile);
      if (await this.pathExists(envPath)) {
        const content = await fs.readFile(envPath, 'utf-8');
        const portMatch = content.match(/PORT\s*=\s*(\d+)/);
        if (portMatch) {
          return parseInt(portMatch[1], 10);
        }
      }
    }

    // 根据框架返回默认端口
    const defaultPorts: Record<string, number> = {
      'Next.js': 3000,
      'React': 3000,
      'Vue.js': 8080,
      'Angular': 4200,
      'Express': 3000,
      'Fastify': 3000,
    };

    return defaultPorts[framework] || 3000;
  }

  private estimateResourceRequirements(framework: string, language: string): ResourceRequirements {
    const requirements: Record<string, ResourceRequirements> = {
      'Next.js': { cpu: '0.5', memory: '512Mi', storage: '1Gi' },
      'React': { cpu: '0.2', memory: '256Mi', storage: '512Mi' },
      'Express': { cpu: '0.3', memory: '256Mi', storage: '512Mi' },
      'Django': { cpu: '0.5', memory: '512Mi', storage: '1Gi' },
      'Flask': { cpu: '0.3', memory: '256Mi', storage: '512Mi' },
      'Spring Boot': { cpu: '0.8', memory: '1Gi', storage: '2Gi' },
    };

    return requirements[framework] || { cpu: '0.5', memory: '512Mi', storage: '1Gi' };
  }
}
```

### 3. 安全增强的AI代理服务

**问题：** 原方案缺少API限流、审计日志和安全验证。

**修正方案：**

```typescript
// packages/server/src/services/ai-proxy.service.ts (安全增强版本)
import { db } from '../db';
import { userAiCredentials, userDeployedProjects, billingRecords, apiQuotas } from '../db/schema/ai-projects';
import { eq, and, gte, sql } from 'drizzle-orm';
import { createCipheriv, createDecipheriv, randomBytes, scryptSync, createHash } from 'crypto';
import { billingQueue } from '../queues/billing-queue';
import { auditLogger } from '../utils/audit-logger';
import { RateLimiter } from './rate-limiter.service';

export class SecureAIProxyService {
  private readonly rateLimiter: RateLimiter;
  private readonly encryptionKey: Buffer;

  constructor() {
    this.rateLimiter = new RateLimiter();
    this.encryptionKey = scryptSync(process.env.CREDENTIAL_ENCRYPTION_KEY!, 'salt', 32);
  }

  async proxyRequest(
    deploymentApiKey: string, 
    requestPayload: any,
    clientIP: string,
    userAgent: string
  ): Promise<any> {
    const requestId = randomBytes(16).toString('hex');
    const startTime = Date.now();

    try {
      // 1. 验证 API Key 并获取项目信息
      const deployedProject = await this.validateApiKey(deploymentApiKey);
      if (!deployedProject) {
        await auditLogger.logSecurityEvent('INVALID_API_KEY', {
          apiKey: deploymentApiKey.substring(0, 8) + '***',
          clientIP,
          userAgent,
        });
        throw new Error('Invalid API Key');
      }

      // 2. 检查速率限制
      const rateLimitKey = `ai_proxy:${deployedProject.userId}`;
      const isAllowed = await this.rateLimiter.checkLimit(rateLimitKey, 100, 3600); // 100 requests per hour
      if (!isAllowed) {
        throw new Error('Rate limit exceeded');
      }

      // 3. 检查配额限制
      await this.checkQuotaLimits(deployedProject.userId, requestPayload);

      // 4. 获取并解密用户AI凭证
      const provider = this.determineProvider(requestPayload);
      const userCreds = await this.getUserCredentials(deployedProject.userId, provider);
      
      // 5. 转发请求到AI服务
      const response = await this.forwardToAIService(userCreds, requestPayload, provider);

      // 6. 记录使用量并更新配额
      await this.recordUsageAndUpdateQuota(deployedProject, response, requestPayload);

      // 7. 记录成功的审计日志
      await auditLogger.logAPIUsage('AI_PROXY_SUCCESS', {
        requestId,
        userId: deployedProject.userId,
        projectId: deployedProject.id,
        provider,
        duration: Date.now() - startTime,
        tokens: response.usage?.total_tokens || 0,
      });

      return response;

    } catch (error) {
      // 记录错误的审计日志
      await auditLogger.logAPIUsage('AI_PROXY_ERROR', {
        requestId,
        error: error.message,
        duration: Date.now() - startTime,
        clientIP,
      });
      
      throw error;
    }
  }

  private async validateApiKey(apiKey: string): Promise<any> {
    const hashedKey = createHash('sha256').update(apiKey).digest('hex');
    
    const project = await db.query.userDeployedProjects.findFirst({
      where: eq(userDeployedProjects.apiKey, hashedKey),
      with: {
        user: true,
      },
    });

    if (project) {
      // 更新最后访问时间
      await db.update(userDeployedProjects)
        .set({ lastAccessedAt: new Date() })
        .where(eq(userDeployedProjects.id, project.id));
    }

    return project;
  }

  private async checkQuotaLimits(userId: string, requestPayload: any): Promise<void> {
    const estimatedTokens = this.estimateTokenUsage(requestPayload);
    
    const quota = await db.query.apiQuotas.findFirst({
      where: and(
        eq(apiQuotas.userId, userId),
        eq(apiQuotas.serviceType, 'AI_TOKENS')
      ),
    });

    if (quota) {
      if (quota.dailyLimit && quota.currentDailyUsage + estimatedTokens > quota.dailyLimit) {
        throw new Error('Daily quota exceeded');
      }
      
      if (quota.monthlyLimit && quota.currentMonthlyUsage + estimatedTokens > quota.monthlyLimit) {
        throw new Error('Monthly quota exceeded');
      }
    }
  }

  private async getUserCredentials(userId: string, provider: string): Promise<string> {
    const creds = await db.query.userAiCredentials.findFirst({
      where: and(
        eq(userAiCredentials.userId, userId),
        eq(userAiCredentials.provider, provider),
        eq(userAiCredentials.isActive, true)
      ),
    });

    if (!creds) {
      throw new Error(`No active ${provider} credentials found`);
    }

    return this.decrypt(creds.encryptedApiKey);
  }

  private async forwardToAIService(apiKey: string, payload: any, provider: string): Promise<any> {
    const endpoints = {
      openai: 'https://api.openai.com/v1/chat/completions',
      anthropic: 'https://api.anthropic.com/v1/messages',
      // 其他提供商...
    };

    const endpoint = endpoints[provider];
    if (!endpoint) {
      throw new Error(`Unsupported AI provider: ${provider}`);
    }

    const response = await fetch(endpoint, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${apiKey}`,
        'Content-Type': 'application/json',
        'User-Agent': 'GitAgent-AI-Proxy/1.0',
      },
      body: JSON.stringify(payload),
      timeout: 30000, // 30 seconds timeout
    });

    if (!response.ok) {
      throw new Error(`AI service error: ${response.status} ${response.statusText}`);
    }

    return await response.json();
  }

  private async recordUsageAndUpdateQuota(
    deployedProject: any,
    response: any,
    requestPayload: any
  ): Promise<void> {
    const usage = response.usage;
    if (!usage) return;

    // 记录详细计费
    const pricing = this.getPricing(requestPayload.model || 'gpt-3.5-turbo');
    const inputCost = (usage.prompt_tokens / 1000) * pricing.input;
    const outputCost = (usage.completion_tokens / 1000) * pricing.output;

    await Promise.all([
      // 记录计费
      db.insert(billingRecords).values([
        {
          userId: deployedProject.userId,
          deployedProjectId: deployedProject.id,
          serviceType: 'AI_TOKENS_INPUT',
          quantity: usage.prompt_tokens.toString(),
          unitPrice: pricing.input.toString(),
          totalCost: inputCost.toString(),
          metadata: { model: requestPayload.model, requestId: response.id },
        },
        {
          userId: deployedProject.userId,
          deployedProjectId: deployedProject.id,
          serviceType: 'AI_TOKENS_OUTPUT',
          quantity: usage.completion_tokens.toString(),
          unitPrice: pricing.output.toString(),
          totalCost: outputCost.toString(),
          metadata: { model: requestPayload.model, requestId: response.id },
        },
      ]),

      // 更新配额使用量
      this.updateQuotaUsage(deployedProject.userId, usage.total_tokens),
    ]);
  }

  private encrypt(text: string): string {
    const iv = randomBytes(16);
    const cipher = createCipheriv('aes-256-cbc', this.encryptionKey, iv);
    let encrypted = cipher.update(text, 'utf8', 'hex');
    encrypted += cipher.final('hex');
    return iv.toString('hex') + ':' + encrypted;
  }

  private decrypt(encryptedText: string): string {
    const [ivHex, encrypted] = encryptedText.split(':');
    const iv = Buffer.from(ivHex, 'hex');
    const decipher = createDecipheriv('aes-256-cbc', this.encryptionKey, iv);
    let decrypted = decipher.update(encrypted, 'hex', 'utf8');
    decrypted += decipher.final('utf8');
    return decrypted;
  }

  private determineProvider(payload: any): string {
    // 根据请求负载确定提供商
    if (payload.model?.startsWith('gpt-')) return 'openai';
    if (payload.model?.startsWith('claude-')) return 'anthropic';
    return 'openai'; // 默认
  }

  private estimateTokenUsage(payload: any): number {
    // 简单的token估算，实际应该更精确
    const text = JSON.stringify(payload);
    return Math.ceil(text.length / 4); // 粗略估算：4个字符≈1个token
  }

  private getPricing(model: string): { input: number; output: number } {
    const pricing = {
      'gpt-4': { input: 0.03, output: 0.06 },
      'gpt-3.5-turbo': { input: 0.001, output: 0.002 },
      'claude-3-opus': { input: 0.015, output: 0.075 },
      // 其他模型...
    };

    return pricing[model] || pricing['gpt-3.5-turbo'];
  }
}
```

### 4. 项目集成服务优化

**问题：** 原方案的集成逻辑缺少错误处理和回滚机制。

**修正方案：**

```typescript
// packages/server/src/services/project-ingestion.service.ts (优化版本)
import { ApplicationService } from './application';
import { GithubService } from './github';
import { EnhancedCodeAnalyzerService } from './code-analyzer.service';
import { db } from '../db';
import { userDeployedProjects } from '../db/schema/ai-projects';
import { randomBytes } from 'crypto';
import * as fs from 'fs/promises';
import * as path from 'path';
import { auditLogger } from '../utils/audit-logger';

export class EnhancedProjectIngestionService {
  constructor(
    private githubService: GithubService,
    private codeAnalyzer: EnhancedCodeAnalyzerService,
    private applicationService: ApplicationService,
  ) {}

  async ingestFromGithub(
    userId: string, 
    githubUrl: string, 
    organizationId: string, 
    projectId: string,
    options: {
      autoStart?: boolean;
      customConfig?: Record<string, any>;
    } = {}
  ): Promise<{
    application: any;
    analysisResult: any;
    deploymentReady: boolean;
  }> {
    const ingestionId = randomBytes(16).toString('hex');
    let repoPath: string | undefined;
    let createdApplicationId: string | undefined;

    try {
      await auditLogger.logActivity('PROJECT_INGESTION_START', {
        ingestionId,
        userId,
        githubUrl,
        organizationId,
        projectId,
      });

      // 1. 验证GitHub URL和访问权限
      await this.validateGithubUrl(githubUrl);

      // 2. 克隆仓库到临时目录
      repoPath = await this.githubService.clonePublicRepo(githubUrl);
      
      // 3. 执行深度代码分析
      const analysisResult = await this.codeAnalyzer.analyze(repoPath);
      
      // 4. 验证分析结果的置信度
      if (analysisResult.confidenceScore < 70) {
        throw new Error(
          `Code analysis confidence too low (${analysisResult.confidenceScore}%). ` +
          'Manual configuration may be required.'
        );
      }

      // 5. 创建Dokploy应用实例
      const applicationData = this.buildApplicationConfig(
        githubUrl, 
        analysisResult, 
        projectId, 
        organizationId,
        options.customConfig
      );

      const newApp = await this.applicationService.createApplication(applicationData, userId);
      createdApplicationId = newApp.id;

      // 6. 生成项目API密钥
      const apiKey = this.generateApiKey();

      // 7. 记录部署项目信息
      await db.insert(userDeployedProjects).values({
        userId,
        dokployApplicationId: newApp.id,
        githubUrl,
        apiKey: this.hashApiKey(apiKey),
        analysisCache: analysisResult,
        deploymentStatus: 'configured',
      });

      // 8. 复制源代码到持久化存储
      const persistentPath = await this.copySourceToPersistentStorage(repoPath, newApp.id);

      // 9. 如果启用自动启动，触发部署
      let deploymentReady = false;
      if (options.autoStart) {
        try {
          await this.applicationService.createDeploymentForImportedApp(newApp.id, userId);
          deploymentReady = true;
        } catch (deployError) {
          console.warn('Auto-deployment failed:', deployError);
          // 不抛出错误，只记录警告
        }
      }

      await auditLogger.logActivity('PROJECT_INGESTION_SUCCESS', {
        ingestionId,
        applicationId: newApp.id,
        analysisScore: analysisResult.confidenceScore,
        deploymentReady,
      });

      return {
        application: { ...newApp, apiKey }, // 只在创建时返回明文API Key
        analysisResult,
        deploymentReady,
      };

    } catch (error) {
      // 回滚操作
      await this.rollbackIngestion(createdApplicationId, repoPath);
      
      await auditLogger.logActivity('PROJECT_INGESTION_FAILED', {
        ingestionId,
        error: error.message,
        userId,
        githubUrl,
      });

      throw error;
    } finally {
      // 清理临时目录
      if (repoPath) {
        await fs.rm(repoPath, { recursive: true, force: true }).catch(console.warn);
      }
    }
  }

  private async validateGithubUrl(githubUrl: string): Promise<void> {
    // 验证URL格式
    const urlPattern = /^https:\/\/github\.com\/[\w-]+\/[\w-]+(?:\.git)?$/;
    if (!urlPattern.test(githubUrl)) {
      throw new Error('Invalid GitHub URL format');
    }

    // 检查仓库是否公开可访问
    const repoPath = githubUrl.replace('https://github.com/', '').replace('.git', '');
    const response = await fetch(`https://api.github.com/repos/${repoPath}`);
    
    if (!response.ok) {
      if (response.status === 404) {
        throw new Error('Repository not found or not public');
      }
      throw new Error(`GitHub API error: ${response.status}`);
    }

    const repoData = await response.json();
    if (repoData.size > 100000) { // 100MB limit
      throw new Error('Repository too large (>100MB)');
    }
  }

  private buildApplicationConfig(
    githubUrl: string,
    analysis: any,
    projectId: string,
    organizationId: string,
    customConfig?: Record<string, any>
  ): any {
    const repoName = githubUrl.split('/').pop()?.replace('.git', '') || 'unknown';
    
    return {
      name: customConfig?.name || repoName,
      description: `Imported from ${githubUrl}`,
      buildType: analysis.buildPack,
      dockerfile: analysis.buildPack === 'dockerfile' ? analysis.dockerfileContent : undefined,
      port: customConfig?.port || analysis.port,
      startCommand: customConfig?.startCommand || analysis.startCommand,
      installCommand: customConfig?.installCommand || analysis.installCommand,
      buildCommand: customConfig?.buildCommand || analysis.buildCommand,
      env: this.formatEnvironmentVariables(analysis.environmentVariables, customConfig?.env),
      cpus: analysis.estimatedResourceRequirements.cpu,
      memory: analysis.estimatedResourceRequirements.memory,
      projectId,
      organizationId,
      // 标记为导入的应用，以便特殊处理
      metadata: {
        importedFrom: githubUrl,
        analysisResult: analysis,
        importedAt: new Date().toISOString(),
      },
    };
  }

  private async copySourceToPersistentStorage(repoPath: string, applicationId: string): Promise<string> {
    const persistentPath = `/dokploy/sources/${applicationId}`;
    await fs.mkdir(persistentPath, { recursive: true });
    await fs.cp(repoPath, persistentPath, { recursive: true });
    return persistentPath;
  }

  private generateApiKey(): string {
    return 'gta_' + randomBytes(32).toString('hex');
  }

  private hashApiKey(apiKey: string): string {
    return require('crypto').createHash('sha256').update(apiKey).digest('hex');
  }

  private formatEnvironmentVariables(
    envVars: any[], 
    customEnv?: Record<string, string>
  ): Record<string, string> {
    const result: Record<string, string> = {};
    
    // 添加分析发现的环境变量（只有默认值的）
    envVars?.forEach(envVar => {
      if (envVar.defaultValue && !envVar.sensitive) {
        result[envVar.name] = envVar.defaultValue;
      }
    });

    // 添加用户自定义的环境变量
    if (customEnv) {
      Object.assign(result, customEnv);
    }

    return result;
  }

  private async rollbackIngestion(applicationId?: string, repoPath?: string): Promise<void> {
    try {
      if (applicationId) {
        // 删除创建的应用
        await this.applicationService.deleteApplication(applicationId);
        
        // 删除部署项目记录
        await db.delete(userDeployedProjects)
          .where(eq(userDeployedProjects.dokployApplicationId, applicationId));
      }
    } catch (rollbackError) {
      console.error('Rollback failed:', rollbackError);
    }
  }
}
```
